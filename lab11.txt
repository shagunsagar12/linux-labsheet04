
mkdir archive_lab

cd ..

ls archive_lab


gunzip archive_lab/file1.txt.gz
ls archive_lab
cat archive_lab/file1.txt


tar -czvf archive.tar.gz archive_lab

tar -tzvf archive.tar.gz


mkdir extracted_tar_gz
tar -xzvf archive.tar.gz -C extracted_tar_gz
ls extracted_tar_gz/archive_lab


zip files.zip archive_lab/file1.txt archive_lab/file2.txt archive_lab/file3.txt

unzip -l files.zip
